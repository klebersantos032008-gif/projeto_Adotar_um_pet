
No JS, dentro de atualizarPets(), conte os pets visíveis:

function atualizarPets() {
    let total = 0

    pets.forEach(function(pet) {
        // ... seu código atual de filtros ...

        if (combinaEspecie && combinaPorte && combinaFavoritos) {
            pet.style.display = ""
            total++
        } else {
            pet.style.display = "none"
        }
    })

    document.querySelector("#status-filtro").textContent =
        total + " pet(s) encontrado(s)"
}
